function exportExcel() { const msg = $("lap-msg"); if (typeof XLSX === "undefined") { msg.textContent = "Library Excel belum termuat. Periksa koneksi internet, lalu muat ulang halaman."; return; } if (!lapData) { msg.textContent = "Tampilkan laporan dulu sebelum diunduh."; return; } msg.textContent = ""; const d = lapData; const itemMap = {}; allItems.forEach((i) => { itemMap[i.id] = i; }); const periodeTeks = tglPanjang(d.start) + " s/d " + tglPanjang(d.end); const waktuEkspor = new Date().toLocaleString("id-ID", { dateStyle: "full", timeStyle: "medium", }); // Sheet 1: mutasi per periode dengan tambahan informasi waktu ekspor const mut = [ ["Laporan Mutasi Stok"], ["Periode", periodeTeks], ["Waktu Ekspor", waktuEkspor], [], ["Kode", "Nama", "Kategori", "Satuan", "Stok Awal", "Masuk", "Keluar", "Stok Akhir"], ]; d.rows.forEach((r) => { mut.push([ r.item.kode, r.item.nama, r.item.kategori || "", r.item.satuan, r.awal, r.masuk, r.keluar, r.akhir, ]); }); const ws1 = XLSX.utils.aoa_to_sheet(mut); ws1["!cols"] = [{ wch: 14 }, { wch: 32 }, { wch: 18 }, { wch: 10 }, { wch: 11 }, { wch: 10 }, { wch: 10 }, { wch: 11 }]; // Sheet 2: stok saat ini const now = [ ["Stok Saat Ini"], ["Waktu Ekspor", waktuEkspor], [], ["Kode", "Nama", "Kategori", "Satuan", "Stok", "Stok Minimum", "Status"] ]; allItems.forEach((i) => { now.push([ i.kode, i.nama, i.kategori || "", i.satuan, Number(i.stok), Number(i.stok_minimum), STATUS_LABEL[statusOf(i)], ]); }); const ws2 = XLSX.utils.aoa_to_sheet(now); ws2["!cols"] = [{ wch: 14 }, { wch: 32 }, { wch: 18 }, { wch: 10 }, { wch: 10 }, { wch: 14 }, { wch: 10 }]; // Sheet 3: riwayat transaksi pada periode const riwayat = [ ["Riwayat Transaksi Periode"], ["Periode", periodeTeks], ["Waktu Ekspor", waktuEkspor], [], ["Waktu", "Kode", "Nama", "Jenis", "Jumlah", "Satuan", "Keterangan", "Petugas"] ]; d.periodTx.forEach((t) => { const item = itemMap[t.item_id]; riwayat.push([ new Date(t.created_at).toLocaleString("id-ID", { dateStyle: "medium", timeStyle: "short" }), item ? item.kode : "", item ? item.nama : "(barang dihapus)", t.jenis === "masuk" ? "Masuk" : "Keluar", Number(t.jumlah), item ? item.satuan : "", t.keterangan || "", d.profilMap[t.user_id] || "", ]); }); const ws3 = XLSX.utils.aoa_to_sheet(riwayat); ws3["!cols"] = [{ wch: 20 }, { wch: 14 }, { wch: 32 }, { wch: 8 }, { wch: 10 }, { wch: 10 }, { wch: 30 }, { wch: 20 }]; const wb = XLSX.utils.book_new(); XLSX.utils.book_append_sheet(wb, ws1, "Mutasi Periode"); XLSX.utils.book_append_sheet(wb, ws2, "Stok Saat Ini"); XLSX.utils.book_append_sheet(wb, ws3, "Riwayat Transaksi"); XLSX.writeFile(wb, "laporan-stok_" + d.dari + "_sd_" + d.sampai + ".xlsx"); toast("File Excel diunduh."); }
